\subsubsection{A strictly stronger radial derivative estimate}
Let
\[
 k(u)=\int_{[0,\infty)}e^{-tu}\,\nu(dt),\qquad
 m_0=\int\nu(dt),\quad m_1=\int t\,\nu(dt)<\infty,
\]
where $\nu$ is positive and finite. Let $Y$ be an integrable circular
complex random variable. Define the real radial coefficient by
\[
 \mathbb E[(w+Y)k(|w+Y|^2)]=w\widetilde k(|w|^2).
\]
The variable $Y$ is independent of the fixed argument and of any driver
to which this convolution will subsequently be applied. No positivity or
complete monotonicity of $\widetilde k$ is assumed.

\begin{theorem}[Averaging the two Bessel bounds before taking moments, \textbf{[P]}]
\label{thm:kcompact-radial}
If $m_0m_1>0$ and $u>0$, then
\[
 \boxed{
 |\widetilde k'(u)|
 \le m_1\Theta\!\left(\frac{um_1}{m_0}\right),\qquad
 \Theta(x)=1-\left(1+\frac1{2x}\right)e^{-1/x}.}
\]
The value at $u=0$ is bounded by $m_1$. If $m_1=0$, the derivative
vanishes. For $x>0$,
\[
 0<\Theta(x)<\min\{1,(2x)^{-1}\}.
\]
Thus the new bound strictly improves
$\min\{m_1,m_0/(2u)\}$ whenever $u,m_0,m_1>0$.
\end{theorem}

\begin{proof}
For a single spectral atom $k(u)=e^{-tu}$, $t>0$, the Fourier calculation
in the incoming proof gives
\[
 \widetilde k'(u)
 =-\frac1{4t}\mathbb E\!\left[
 R^4\frac{J_2(R\sqrt u)}{R^2u}\chi_Y(R)\right],
 \qquad \frac{R^2}{4t}\sim\operatorname{Exp}(1),\quad |\chi_Y|\le1.
\]
The two global bounds $|J_2(x)|\le x^2/8$ and $|J_2(x)|\le1/2$
may be imposed inside the \emph{same} expectation. Writing $V\sim
\operatorname{Exp}(1)$ and $a=1/(tu)$ gives
\[
 |\widetilde k'(u)|
 \le \frac t2\mathbb E\min\{V^2,aV\}
 =t\left[1-\left(1+\frac a2\right)e^{-a}\right]=:B_u(t).
\]
Indeed,
\[
 \int_0^a v^2e^{-v}\,dv+a\int_a^\infty ve^{-v}\,dv
 =2-(a+2)e^{-a}.
\]
This expectation is strictly smaller than either untruncated moment,
which proves the strict comparison for a single atom.
For fixed $u>0$, extend $B_u$ continuously by $B_u(0)=0$. Direct
differentiation gives
\[
 B_u'(t)=1-e^{-a}(1+a+a^2/2)\ge0,\qquad
 B_u''(t)=-\frac{a^3e^{-a}}{2t}\le0.
\]
The triangle inequality followed by Jensen for the positive measure
$\nu/m_0$ now yields
\[
 |\widetilde k'(u)|\le\int B_u(t)\,\nu(dt)
 \le m_0B_u(m_1/m_0)=m_1\Theta(um_1/m_0).
\]
The bound $B_u(t)\le t$ justifies differentiation and integration
by domination. A spectral atom at zero contributes a constant vector
coefficient and no derivative, since $\mathbb E Y=0$.
\end{proof}

For example, at $um_1/m_0=1/2$ the old bound is $m_1$, whereas the
new bound is $(1-2e^{-2})m_1<0.73m_1$. This is an improvement to an
intermediate estimate, not a percentage improvement to $K_G^{\mathbb C}$.

\paragraph{Rational facets used by the verifier.}
For any $a>0$, concavity of $B_u(t)$ gives the tangent inequality
\[
 \Theta(x)\le A_a+\frac{B_a}{2x},\qquad
 A_a=1-(1+a+a^2/2)e^{-a},\quad B_a=(1+a)e^{-a}.
\]
We use $a\in\{1/4,1/2,1,2,4,8\}$, rounding both $A_a$ and $B_a$
\emph{upward}. Taking a minimum with $1$ and $(2x)^{-1}$ is valid.
No discretization of an unknown spectral measure is involved.

\subsubsection{The content formula for any finite number of drivers}
Let $Z,X_1,\ldots,X_N,U$ be independent proper standard complex
Gaussians, let $c_j,g\ge0$, and let $\alpha_j,\beta_j\in\mathbb C$.
Put
\[
 H(X)=X(|X|^2-2)/\sqrt2,\quad T=|Z|^2,\quad
 q_0=\sum_{j=1}^N c_jH(X_j),\quad v=g^2,
\]
\[
 G=q_0+\sqrt vU+(T-1)\sum_j\alpha_jX_j+
       \frac{Z^2}{\sqrt2}\sum_j\beta_j\overline X_j,\qquad
 A=\left(\sum_j|\alpha_j|^2\right)^{1/2}.
\]
These are independent remote driver blocks with shared A/B inside each
block. A general tensor containing $X_iX_j\overline X_k$ is not assumed
to admit this diagonal form.

For the positive buffered kernels $k_{j,v}$, $j=A,B$, write
$m_{j,0}=k_{j,v}(0)$ and $m_{j,1}=-k_{j,v}'(0)$. Set
\[
 Z_3(t)=t(t-2)^2/2,\qquad
 H_*(s)=\int_{\{Z_3(t)>s\}}t(t-2)e^{-t}\,dt.
\]
The signed-tail lemma gives $H_*\ge0$ and $\int H_*=16$.
Define
\[
 d_j^\sharp(c,v)=\frac{c^3m_{j,1}}{\sqrt2}
       \int_0^\infty H_*(s)
       \Theta\!\left(\frac{c^2s m_{j,1}}{m_{j,0}}\right)\,ds,
 \qquad D_j^\sharp=\left(\sum_kd_j^\sharp(c_k,v)^2\right)^{1/2}.
\]
At $c=0$ define $d_j^\sharp=0$.
Then the actual contents obey $|d_{j,k}|\le d_j^\sharp(c_k,v)$.
To prove this, integrate out the buffer and all other independent
circular driver terms, use the signed-tail identity, and apply
Theorem~\ref{thm:kcompact-radial}. The same argument proves the
older cubic and saturated bounds by replacing $\Theta$ by $1$ or
$\min\{1,(2x)^{-1}\}$, respectively.

The bound increases in each of $c,m_{j,0},m_{j,1}$ separately.
For $m_0,m_1$ this follows from the positive, concave perspective
$m_0B_u(m_1/m_0)$. For $c$ write its integrand, apart from a positive
constant, as
\[
 c^3\Theta(Kc^2)
 =\frac12\mathbb E\min\{c^3V^2,cV/K\},
 \qquad K>0.
\]
Thus upper moment bounds and upper amplitude endpoints may be inserted
before interval integration.

\subsubsection{The common remote moment bound and simultaneous insertion}
Write
\[
 e_0=\sum_jc_j^2,\quad k_4=\sum_jc_j^4,\quad k_6=\sum_jc_j^6.
\]
The exact moments, for every finite $N$, are

\]
For $q=q_0+\sqrt vU$, replace $e_0$ by $e=e_0+v$ in these two
polynomials, leaving $k_4,k_6$ unchanged. These identities follow
by independence, circularity, and the single-driver moments
$1,44,11352$. They do not need a separate two-driver proof.

Let $h(v)=2\mathbb E|L+\sqrt vU|$, $F(v)=h(v)-pv$,
$C(v)=-h''(v)$ and $B(v)=h'''(v)/(3C(v))$. Three valid upper bounds
on $\Lambda(q)-S$ are:
\[
 R_1=F(v)-S+(h'(v)-p)e_0-\mathcal L,\quad
 R_2=F(v+e_0)-S+C(v)e_0^2/2-\mathcal L,

\[
 \mathcal L=\frac{C(v)}2
       \frac{M_{4,0}^2}{2M_{4,0}+B(v)M_{6,0}},
\]
with $\mathcal L=0$ at $e_0=0$, and the anchored Hermite bound
\[
 R_H=p_0+(p_1-p)e+p_2(2e^2+42k_4)
       +p_3(6e^3+378ek_4+10968k_6)-\Delta_0.
\]
Here $P_{x,y}(u)=p_0+p_1u+p_2u^2+p_3u^3$ is any certified
global cubic majorant for
$2\mathbb E|L+\sqrt u|-h(0)$, and $\Delta_0\le S-h(0)$.
The spectral Radau plus cubic Hermite construction in the incoming
note supplies such majorants. It bounds a fixed shift; it must not
be replaced by the different function $h(u)$ for added Gaussian variance.

\begin{theorem}[Master independent-driver envelope, \textbf{[P]}]
\label{thm:kcompact-envelope}
With $R=\min\{R_1,R_2,R_H\}$, omitting any unavailable candidate,
\[
 \boxed{\Lambda(G)-S\le
 R+2D_A^\sharp A+
 \left(\frac{207}{100}-p\right)A^2+
 \frac{\left(D_B^\sharp+\frac{19}{25}A\right)^2}{369/1000}.}
\]
\end{theorem}
\begin{proof}
Insert both sectors simultaneously at the $Z$-independent baseline
$q$. The accepted potentials and the exact orthogonal variance price give,
with $D=\|\beta\|_2$,
\[
 \Lambda(G)-\Lambda(q)
 \le 2D_A^\sharp A+2D_B^\sharp D+
 (207/100-p)A^2+2(19/25)AD-(369/1000)D^2.
\]
Complete the last square over $D\ge0$. No independence between a
driver's cubic and its A/B terms has been introduced.
\end{proof}

\paragraph{How to use the formula.}

does not say that its right side is nonpositive everywhere. The
certificates below concern $N=2$. They do not cover extra Y/W/diagonal
terms or general mixed cubic monomials. Those have separate certificates.

\subsubsection{A joint projection estimate}
Let $q$ be independent of $Z$ and square-integrable, with enough moments
for the expressions below. Let $\mathcal D$ be the span of the
orthonormal variables $X_1,\ldots,X_N$ in ordinary complex $L^2$.
Suppose $P_{\mathcal D}q=0$. Define
\[
u=|q|^2,\quad k=k_B,\quad
d=P_{\mathcal D}[qk(u)],\quad v=P_{\mathcal D}[qu].
\]
The visible kernel identity is
\[
\E_Z\frac{Z^2}{\sqrt2}\frac{\overline{L+w}}{|L+w|}
=-wk_B(|w|^2),\qquad k_B(u)=\int_0^\infty e^{-tu}\nu_B(\,d t),\quad\nu_B\ge0.
\]
Thus $\norm d$ is precisely the norm of the B-content coefficient
vector, up to an irrelevant sign.

\begin{lemma}[Affine projection subtraction, {\normalfont\textbf{[P]}}]
For any real $\lambda_0,\lambda_1$, set
$r(u)=k(u)-\lambda_0-\lambda_1u$. Then
\[
d=\lambda_1v+P_{\mathcal D}[qr(u)],\qquad
\norm{P_{\mathcal D}[qr(u)]}_2^2\le\E[u r(u)^2].
\]
\end{lemma}
\begin{proof}
The first equality uses $P_{\mathcal D}q=0$. The inequality is Bessel's
inequality in unweighted $L^2$.
\end{proof}
This does not replace shared drivers by independent copies. It uses no
weighted conditional projection inequality.

Write
\[
\mathfrak b=369/1000,\qquad \chi=19/25,\qquad
\mathfrak a=207/100-p.
\]
The simultaneous all-shift potential estimate, with a baseline
independent of $Z$, gives
\begin{equation}\label{kdual-potential}
\Lambda(G)-S\le\Lambda(q)-S+2D_AA+\mathfrak a A^2
+2B\norm d+2\chi AB-\mathfrak b B^2,
\end{equation}
where $D_A$ is any upper bound for the norm of the A-content vector.
To see the constants, insert both sectors together in
\[
2|F+J|\le2|F|+2\operatorname{Re}(\bar FJ)/|F|+|J|^2/|F|,
\]
condition on the remote variables, and then apply the potentials in
$Z$. They give $C_A<207/100$, $C_{AB}<19/25$ and $C_B<953/1000$;
$p-C_B>369/1000$. Cauchy--Schwarz bounds the cross term by $2\chi AB$.
The transformation $Z\mapsto-Z$ changes $L$ to $-L$ and leaves $G$
unchanged, so the two absolute-value expectations are equal.
Orthogonality gives the variance price exactly.

\begin{proposition}[Joint scalar dual, {\normalfont\textbf{[P]}}]\label{kdual-dual}
Let $b_1,b_2>0$ with $b_1+b_2=\mathfrak b$, and let a cubic polynomial
$P(u)=\sum_{j=0}^3p_ju^j$ satisfy, for every $u\ge0$,
\begin{equation}\label{kdual-scalar}
f(u)-h(0)+\frac{u}{b_2}(k_B(u)-\lambda_0-\lambda_1u)^2\le P(u),
\end{equation}
where $f(u)=2\E|L+\sqrt u|$ and $h(v)=2\E|L+\sqrt vU|$.
If $q$ is circular and $\Delta\le S-h(0)$, then
\begin{equation}\label{kdual-dualbound}
\Lambda(G)-S\le\E P(|q|^2)-p\E|q|^2-\Delta
+2D_AA+\mathfrak a A^2+
\frac{(|\lambda_1|\norm v+\chi A)^2}{b_1}.
\end{equation}
\end{proposition}
\begin{proof}
The lemma and Young's inequality give
\[
2B\norm d\le2B|\lambda_1|\norm v+b_2B^2+\E[u r(u)^2]/b_2.
\]
Apply this in \eqref{kdual-potential} and maximize the remaining quadratic
$-b_1B^2+2B(|\lambda_1|\norm v+\chi A)$ over $B\ge0$.
Circularity and independence give $\Lambda(q)=\E f(|q|^2)-p\E|q|^2$.
Finally use \eqref{kdual-scalar} and $S-h(0)\ge\Delta$.
\end{proof}

\subsubsection{Self-driver moments and the A estimate}

\E[\bar X_jq|q|^2]&=8\sqrt2c_j^3,\qquad
\norm v^2=128k_6.\label{kdual-projection}
\end{align}
For completeness,
\[
\E[t(t-2)]=0,\quad \E[t(t-2)^2]=2,\quad
\E[t^2(t-2)^4]=176,\quad \E[t^3(t-2)^6]=90816,

\begin{theorem}[All independent Wick B-only sums, \textbf{[P,C]}]
\label{thm:k-bonly-all}
At $a=699/1250$, $b=179/1250$, $p=2500/1891$, let
$Z,X_1,\ldots,X_N,U$ be independent proper standard complex Gaussians,
$H(X)=X(|X|^2-2)/\sqrt2$, and
\[
 G=\sum_{j=1}^N\left[c_jH(X_j)+\beta_jZ^2\overline X_j/\sqrt2\right]+gU.
\]
For every finite $N$, every $c_j,g\ge0$, and every complex $\beta_j$,
\[
 \boxed{\Lambda(G)<S+\frac{75}{10^6}.}
\]
\end{theorem}

\begin{proof}
Write $q=\sum_jc_jH(X_j)+gU$, $e=\sum_jc_j^2+g^2$,
$k_r=\sum_jc_j^r$, and $\mathfrak b=369/1000$.
The exact independent-driver identities are
\[
 M_4=\E|q|^4=2e^2+42k_4,\qquad
 M_6=\E|q|^6=6e^3+378ek_4+10968k_6,
\]
\[
 P_{\mathcal D}q=0,\qquad
 \|P_{\mathcal D}(q|q|^2)\|_2^2=128k_6,
 \quad\mathcal D=\operatorname{span}(X_1,\ldots,X_N).
\]
The joint scalar-dual interface therefore gives, for every certified row,
\begin{equation}\label{eq:k-bonly-all-row}
 \Lambda(G)-S\le
 P_0+(P_1-p)e+P_2M_4+P_3M_6
 +\frac{128\lambda_1^2}{b_1}k_6-\Delta,
\end{equation}
where $b_1+b_2=\mathfrak b$, $b_1,b_2>0$, $P_3\ge0$,
$\Delta\le S-h(0)$, and the required scalar inequality is
\[
 f(u)-h(0)+\frac{u}{b_2}\bigl(k_B(u)-\lambda_0-\lambda_1u\bigr)^2
 \le\sum_{r=0}^3P_ru^r\qquad(u\ge0).
\]
Here $f(u)=2\E|L+\sqrt u|$ is the fixed-shift function.
The inherited directed benchmark gives $\Delta>0.0020708$.

To certify new rows, extend the same positive-measure moment integrator to
nine spectral slope moments and nine B-kernel moments.
For $H_j=(-1)^jh^{(j+1)}(0)$ the slope moments are $H_j/(j+1)!$.
The higher B coefficients continue the inherited formula
\[
 m_{B,j}=\frac{1}{2\sqrt2}
 \frac{\binom{2j+2}{j+1}}{4^{j+1}}R_{j-1}^{3,1}
 \quad(1\le j\le8),
\]
where $R_{j-1}^{3,1}$ is the raw integral of the preceding moment
recurrence; $m_{B,0}$ keeps its separate integral.
Composite Gauss--8 on $[0,8]$, mesh $1/32$, uses the same Cauchy
remainder and Gaussian tail estimates, added outward.

Use zero-Radau--5 for the slope measure and Gauss--4/zero-Radau--5
for the B measure. They give $f(u)-h(0)\le F_+(u)$ and
$K_-(u)\le k_B(u)\le K_+(u)$ on the whole half-line.
Indeed, the eighth derivative of $e^{-tu}$ in $t$ is nonnegative,
while its ninth derivative is nonpositive and the Radau remainder
factor is $t\prod_{i=1}^4(t-x_i)^2\ge0$.
For the slope rule apply this also to
$(1-e^{-tu})/t=\int_0^u e^{-ts}\,ds$.
The exact verifier solves the Hankel systems with outward intervals,
isolates all positive nodes by disjoint exact sign brackets, and
bounds positive weights through the Lagrange formulas.
Floating root guesses cannot accept a bracket.

Forty rational rows are verified against both $K_-$ and $K_+$.
Convexity of the squared residual gives the true-kernel inequality.
Directed compact cells and analytic cubic tails from $u=128$ prove
all forty rows globally, in $8\,659$ accepted cells.

Set $x=\sqrt{k_4}$ and $w=e-x\ge0$. Since $k_6\le x^3$ and
$10968P_3+128\lambda_1^2/b_1\ge0$, substituting $k_6=x^3$ in
\eqref{eq:k-bonly-all-row} gives an upper bound. The new directed cover of
\[

\]
accepts $348$ closed boxes, with largest outward upper bound

Here $w$ is only a moment parameter; it is not a Gaussian variance

For $x\le1/180$, Theorem~\ref{thm:bonly-diffuse} gives
\[

 =\frac{4891}{66600000}<\frac{75}{10^6}.
\]
For $x\ge3/100$, Theorem~\ref{thm:k-joint-dual-many} applies with
$A=0$. Its 87 scalar rows and arbitrary-number cover are included
in the replay. All joining endpoints are included.
Finally, if $e\ge1$, then $V=\E|G|^2\ge1$ and orthogonality plus
Jensen give
\[
 \Lambda(G)\le2\sqrt{1181331/1562500+V}-pV<4/3<S.
\]
The function decreases for $V\ge1$; rational squaring checks its
value at $1$. This completes every variance and buffer.
\end{proof}

\begin{corollary}[A small global A strip, \textbf{[P,C]}]
With the same independent drivers, let
$G_A=G+(|Z|^2-1)r_A$, where $r_A=\sum_j\alpha_jX_j$ and
$A=(\sum_j|\alpha_j|^2)^{1/2}$. Then
\[
 \Lambda(G_A)<S+\frac{75}{10^6}+\frac43 A-pA^2.
\]
In particular, $A\le1/100000$ implies $\Lambda(G_A)<S+89/10^6$,
with no restriction on $N,c_j,g$, or the complex B coefficients.
\end{corollary}
\begin{proof}
The added sector is orthogonal to $G$ by $\E(|Z|^2-1)=0$ and
visible-phase cancellation. The triangle inequality therefore gives
\[
 \Lambda(G_A)-\Lambda(G)
 \le2\E\bigl|(|Z|^2-1)r_A\bigr|-pA^2
 =\frac{2\sqrt\pi}{\mathrm e}A-pA^2\le\frac43A-pA^2.
\]
Here $r_A$ is proper Gaussian with second moment $A^2$ and independent
of $Z$. Use $\E||Z|^2-1|=2/\mathrm e$,
$\sqrt\pi<16/9$ and $\mathrm e>8/3$. Finally
$75/10^6+4/(3\cdot100000)<89/10^6$.
For $A=0$, strictness comes from the preceding theorem.
\end{proof}

\subsection{A shared \(B\) theorem for arbitrary coupled remote cubics}
\label{sec:general-coupled-B-cumulant}

\paragraph{Status and scope [P,C].}
Keep the peak parameters \(p=2500/1891\) and
\(L=Z(699/1250+(179/1250)|Z|^2)\).
Let \(q\) be an arbitrary finite-dimensional remote \(H_{2,1}\) cubic,
let \(U\) be a fresh independent proper Gaussian, and let \(R\) be an
antiholomorphic Gaussian linear form in the original remote coordinates.
The cubic and \(R\) may share every driver. They are independent of
the visible \(Z\), and \(R\) does not use the fresh \(U\).
Put
\[
 Q=q+gU,\quad g\ge0,\qquad e=\mathbb E|Q|^2,\quad
 \kappa=\mathbb E|q|^4-2(\mathbb E|q|^2)^2,\quad B=\|R\|_2.
\]
Under the accepted sixth-moment and contraction inputs, the new bounds are
\begin{align}
 \kappa\ge\frac1{100}
 &\ \Longrightarrow\
 \Lambda(Q+Z^2R/\sqrt2)
 <S-\frac1{50000}-\frac{B^2}{2000},                 \label{eq:general-B-strict}\\
 \kappa\ge\frac9{2500}
 &\ \Longrightarrow\
 \Lambda(Q+Z^2R/\sqrt2)
 <S+\frac{89}{10^6}-\frac{B^2}{2000}.               \label{eq:general-B-weak}
\end{align}
These conditions concern the \emph{unnormalized} cumulant.
There is no independent-Wick assumption on \(q\), and no amplitude
restriction on \(R\) or \(g\). No additional visible sector is included.

\paragraph{Inputs retained explicitly.}
Write \(x=\sqrt\kappa\). The existing moment input is
\[
 M_4=2e^2+x^2,\qquad
 M_6\le6e^3+9ex^2+\frac{4045313}{100000}x^3,
 \qquad x\le\frac{31}{4}e.
\]
The coefficient \(4045313/100000\) remains an analytic input, not a
consequence of the numerical replay here.
Let \(\mathcal D\) be the holomorphic first-chaos space of the original
remote coordinates, excluding \(U\), and put
\(V=\|P_{\mathcal D}(Q|Q|^2)\|_2^2\).
Then \(P_{\mathcal D}Q=0\), and the accepted contraction bound gives
\[
 V\le\frac{2203}{1000}x^3.
\]
Indeed the sharper accepted constant is
\(\sqrt{142688/29403}<2203/1000\).
Adding the fresh buffer changes neither \(\kappa\) nor this projection:
\(\mathbb E_U[Q|Q|^2]=q|q|^2+2g^2q\).

\paragraph{A smooth relaxed contraction bound [P].}
The homogeneous directional contraction estimate has the form
\[
 V\le(e_0+d+f)(\kappa-2d^2-2f^2),\qquad
 e_0=\mathbb E|q|^2,\quad d,f\ge0.
\]
The nonnegative budget and \(2d^2+2f^2\ge(d+f)^2\) imply
\[
 V\le\max_{0\le s\le\sqrt\kappa}(e_0+s)(\kappa-s^2)
 \le H(e,\kappa),
 \qquad
 H(e,\kappa)=(e+s)(\kappa-s^2),\quad
 s=\frac{\sqrt{e^2+3\kappa}-e}{3}.
\]
The last substitution uses \(e\ge e_0\) and monotonicity in \(e\).
The envelope derivatives are particularly simple:
\[
 \partial_eH=\kappa-s^2,\qquad
 \partial_xH(e,x^2)=2x(e+s).
\]
They allow a directed cover without division by \(e\) or a split at a
normalized-kurtosis threshold.

\paragraph{The scalar dual [P].}
For any of the certified rows
\((\lambda_0,\lambda_1,b_2,P)\), write

\begin{align*}
 \Lambda(Q+Z^2R/\sqrt2)+\frac{B^2}{2000}-S
 \le{}&P_0+(P_1-p)e+P_2(2e^2+x^2)\\
 &+P_3\left(6e^3+9ex^2+\frac{4045313}{100000}x^3\right)\\

       \min\left\{\frac{2203}{1000}x^3,H(e,x^2)\right\}
       -\Delta .
\end{align*}
Here \(\Delta\) is the existing certified lower bound for \(S-f(0)\);
the replay proves \(\Delta>20708/10^7\).
Every \(P_3\) is nonnegative, justifying the sixth-moment substitution.
This argument requires only independence from \(Z\), not independence
between \(q\) and the shared linear driver.

\paragraph{The full cover [C].}
