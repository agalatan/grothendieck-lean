\subsection{A sharp lower bound for Gaussian smoothing}
\begin{lemma}[Inverse Rice mean; P]
\label{lem:inverse-rice-pi}
For a standard proper complex Gaussian \(Y\), \(D\ge0\), and \(v>0\),
\[
\boxed{\E|\sqrt D+\sqrt vY|^{-1}
\ge(D+v/\pi)^{-1/2}.}
\]
The constant \(\pi\) is the largest possible constant in this form.
\end{lemma}
\begin{proof}
Gaussian integration of the positive Laplace formula for the reciprocal
norm gives
\[
\begin{aligned}
\E|\sqrt D+\sqrt vY|^{-1}
&=\frac1{\sqrt\pi}\int_0^\infty
\frac{t^{-1/2}}{1+vt}
\exp\left(-\frac{Dt}{1+vt}\right)\,dt\\
&=\sqrt{\frac\pi v}\,\E e^{-(D/v)B},
\end{aligned}
\]
where \(B\sim\operatorname{Beta}(1/2,1/2)\).
Its density is \(1/(\pi\sqrt{t(1-t)})\) on \((0,1)\).
The density of \(A\sim\operatorname{Gamma}(1/2,\text{scale }\pi)\)
is \(e^{-t/\pi}/(\pi\sqrt t)\).  On \((0,1)\) the former dominates
the latter.  Both have mass one, so \(B\) is stochastically smaller
than \(A\).  Hence
\[
\E e^{-yB}\ge\E e^{-yA}=(1+\pi y)^{-1/2}.
\]
This proves the bound.  At \(D=0\) there is equality, proving optimality.
\end{proof}

\begin{theorem}[Conditional removal of Gaussian buffer; P]
\label{thm:conditional-heat-pi}
Let \(F\in L^2\) be complex, let \(\mathcal A\) be a sigma-field, and
let \(Y\) be independent of \(\sigma(F,\mathcal A)\).
Put \(D=\E(|F|^2\mid\mathcal A)\).  For \(0\le g\le h\),
\[
\boxed{
2\E(|F+hY|-|F+gY|\mid\mathcal A)
\ge\pi\left(\sqrt{D+h^2/\pi}-\sqrt{D+g^2/\pi}\right).}
\]
\end{theorem}
\begin{proof}
For positive variance, the heat identity is
\[
\frac{d}{dv}\,2\E|F+\sqrt vY|
=\frac12\E|F+\sqrt vY|^{-1}.
\]
Apply Lemma~\ref{lem:inverse-rice-pi} conditional on \(F\), then
conditional Jensen to the convex function
\(u\mapsto(u+v/\pi)^{-1/2}\).  Integrating over
\([g^2,h^2]\) proves the assertion.  At \(g=0\), pass to the limit
in \(L^1\).  No negative moment of \(F\) is required.
\end{proof}
For any \(0<C\le\pi\) one may replace the right side by
\[
f_{C,h,g}(D)=
C\left(\sqrt{D+h^2/C}-\sqrt{D+g^2/C}\right)
=\frac{h^2-g^2}
{\sqrt{D+h^2/C}+\sqrt{D+g^2/C}}.
\]
The last expression increases with \(C\).
The old conditional Jensen comparison is \(C=1\); the first new
directed continuation uses \(C=3\).

\subsubsection{A stronger region without an asymptotic remainder}
\begin{lemma}[Second-term inverse Rice minorant; P]
\label{lem:rice-second-term}
If \(D\ge2v>0\), then
\[
\E|\sqrt D+\sqrt vY|^{-1}
\ge D^{-1/2}\left(1+\frac{v}{4D}\right).
\]
If \(D\ge v>0\), the weaker lower bound \(D^{-1/2}\) holds.
\end{lemma}
\begin{proof}
Compare the beta density with \((1+t/2)/(\pi\sqrt t)\) on the
whole positive half-line.  Their difference is positive on \((0,1)\)
and negative on \((1,\infty)\).  The second density has Laplace
transform \((\pi y)^{-1/2}(1+1/(4y))\).
At \(y=2\), the beta transform is
\[
e^{-1}I_0(1)\ge\frac5{4e}>
\frac5{11}>\frac9{20}>\frac9{8\sqrt{2\pi}},
\]
using \(e<11/4\), \(\pi>25/8\), and the first two terms
\(I_0(1)\ge1+1/4\).
For every \(y\ge2\), weighting this signed difference by
\(e^{-(y-2)t}\) multiplies its positive part by at least
\(e^{-(y-2)}\), and its negative part by at most that amount.
Its positive Laplace transform therefore remains positive.
The beta representation in Lemma~\ref{lem:inverse-rice-pi} proves
the first claim.  For the weaker claim, compare instead with
\(1/(\pi\sqrt t)\); at \(y=1\), Jensen gives
\(\E e^{-B}\ge e^{-1/2}>1/\sqrt\pi\).
\end{proof}
The reciprocal Rice mean is completely monotone in \(D\).
Conditional Jensen can therefore be applied before
Lemma~\ref{lem:rice-second-term}.  If
\(D=\E(|F|^2\mid\mathcal A)\ge2h^2\), integration gives
\begin{equation}
2\E(|F+hY|-|F+gY|\mid\mathcal A)
\ge \frac{h^2-g^2}{2\sqrt D}
 +\frac{h^4-g^4}{16D^{3/2}}.
\label{eq:rice-heat-two-term}
\end{equation}
The condition is on the conditional mean, not every realization of \(F\).

\paragraph{The rearrangement estimate [P].}
Set \(P(s)=s(a+bs)^2\).  The pushforward under \(z\mapsto L\) of Gaussian
measure weighted by \(|z|^4/2\) has planar density \(f_B(|w|^2)/\pi\), where
\begin{equation}
 f_B(P(s))=\frac{s^2e^{-s}}{2(a+bs)(a+3bs)}.
 \label{eq:b-rearrange-density}
\end{equation}
For any measurable planar set \(E\) of area \(\pi A\), and any \(u\),
layer cake gives
\[
 \frac1\pi\int_E\frac{dw}{|w-u|}
 \leq\frac1\pi\int_0^\infty\min(\pi A,\pi/t^2)\,dt=2\sqrt A.
\]
Suppose a nonnegative density numerator is bounded on disjoint annuli
by heights \(b_i\), with respective areas divided by \(\pi\) at most
\(a_i\).  Sort the heights in decreasing order and set
\(A_i=\sum_{j\leq i}a_j\), \(A_0=0\).  Applying the preceding inequality
to all superlevel sets yields
\begin{equation}
 \sup_u\int\frac{f_{\rm step}(|w|^2)}{\pi|w-u|}\,dw
 \leq 2\sum_i b_i(\sqrt{A_i}-\sqrt{A_{i-1}}).
 \label{eq:b-rearrange-step}
\end{equation}
Inflating any \(a_i\) increases every superlevel area, hence increases
this bound.  This justifies outward area bounds despite the differences
of square roots in the displayed expression.  A remaining density with
mass \(m\) and height at most \(M/\pi\) contributes at most \(2\sqrt{mM}\).

\subsection{Positive content for radial cubics in every dimension}
\paragraph{Status [P]; exact polynomial audit [C].}
Let $X=(X_1,\ldots,X_n)$ be standard proper complex Gaussian,
$m=n+1$, and
\[
 Q_n=\frac{X_1(\|X\|^2-m)}{\sqrt m}\in\mathcal H_{2,1},
 \qquad \|Q_n\|_2=1.
\]
Let the visible $Z$, the fresh Gaussian $W$, and a nonnegative random
variance $V$ be independent of each other and of $X$, with
$\mathbb EV<\infty$. Put $U_c=cQ_n+\sqrt VW$, $c\ge0$.
The averaged kernels of the preceding zero-buffer theorem are
$C_A(w)=-\bar w k_A(|w|^2)$ and $T_B(w)=-w k_B(|w|^2)$,
with $k_A,k_B$ nonnegative completely monotone.
For either content $d_j$, $j=A,B$, in the shared first-coordinate
direction,
\begin{equation}
 \boxed{\frac{d_j(c)}{c^3}\text{ is nonnegative completely monotone
 in }c^2,\qquad
 d_j(c)\le
 \frac{2(5m+6)}{m\sqrt m}[-k_j'(0)]c^3.}
 \label{eq:radial-content-cm}
\end{equation}
Every content component in a direction orthogonal to $X_1$ is zero.
The multiplier equals $8\sqrt2$ for $n=1$ and decreases as
$10/\sqrt m$.

\paragraph{The Gamma-order lemma.}
For any real $m>0$, let $R\sim\operatorname{Gamma}(m)$ and
$Z_m(r)=r(r-m)^2/m$. Then
\[
 J_m(x)=\mathbb E[(R-m)e^{-xZ_m(R)}],\qquad
 -J_m(x)/x=\int_0^\infty e^{-xu}H_m(u)\,du,
 \quad H_m\ge0,\quad \int H_m=5m+6.
\]
Indeed $-(r^me^{-r})/\Gamma(m)$ is an antiderivative of
$(r-m)r^{m-1}e^{-r}/\Gamma(m)$.
For $0<u<4m^2/27$, write the three roots of $Z_m(r)=u$ as
$r_1<m/3<r_2<m<r_3$. They sum to $2m$, and the signed tail is
\[
 H_m(u)=
 \frac{r_3^me^{-r_3}-r_2^me^{-r_2}+r_1^me^{-r_1}}{\Gamma(m)}.
\]
Since
\[
 \log(r_3/r_2)\ge\frac{2(r_3-r_2)}{r_2+r_3}
 \ge\frac{r_3-r_2}{m},
\]
the first term dominates the second. Above the critical value the
negative tail is empty. Integration by parts gives the Laplace formula,
and Gamma moments give
$\int H_m=\mathbb E[R(R-m)^3]/m=5m+6$.
This proves the lemma for all shapes without a numerical root argument.

\paragraph{Size bias and content.}
For $n>1$, $S=\|X\|^2$ and $B=|X_1|^2/S$ are independent
$\operatorname{Gamma}(n)$ and $\operatorname{Beta}(1,n-1)$ variables.
Size bias by $SB$, of expectation one, changes them to independent
$R\sim\operatorname{Gamma}(m)$ and
$B_*\sim\operatorname{Beta}(2,m-2)$, with $\mathbb EB_*=2/m$.
For $m=2$ take $B_*=1$.
Thus
\[

\subsection{A proved moment refinement for the next continuation}
\paragraph{Status [P]; no extra numerical region claimed.}
For $0\le g<h$ and $C>0$, the heat minorant is completely monotone:
\[
 f_{C,h,g}(u)=\frac{C}{2\sqrt\pi}\int_0^\infty e^{-ut}
 (e^{-g^2t/C}-e^{-h^2t/C})t^{-3/2}\,dt.
\]
Let $U\ge0$ have moments $m_j=\E U^j$, $j\le3$, and suppose
$b_1=m_2-m_1^2>0$. Put
\[
 a_1=\frac{m_3-2m_1m_2+m_1^3}{b_1},\qquad
 J=\begin{pmatrix}m_1&\sqrt{b_1}\\ \sqrt{b_1}&a_1\end{pmatrix}.
\]
The matrix is positive semidefinite since
\[
 a_1=\frac{\E[U(U-m_1)^2]}{b_1}\ge0,\qquad
 \det J=\frac{m_1m_3-m_2^2}{b_1}\ge0.
\]
Let $x_\pm$ be its eigenvalues and $w_\pm$ the weights matching
mass one and mean $m_1$. Then
\[
 \boxed{\E f_{C,h,g}(U)\ge
 w_-f_{C,h,g}(x_-)+w_+f_{C,h,g}(x_+)
 \ge f_{C,h,g}(m_1).}
\]
The two-node rule matches moments through degree three. Cubic Hermite
interpolation at the two positive nodes is a global minorant, because
$f^{(4)}\ge0$ and its remainder contains
$(u-x_-)^2(u-x_+)^2$. If a node is zero, the equality case of
$m_1m_3\ge m_2^2$ forces support on zero and one positive point, so
the rule is exact. If $b_1=0$, use $f(m_1)$ with equality.
This can be applied after conditioning on one or both radii. The
companion Markdown proof gives exact Fourier formulas for the three
moments with general relative phases. No numerical gain from this
refinement has been claimed here.
