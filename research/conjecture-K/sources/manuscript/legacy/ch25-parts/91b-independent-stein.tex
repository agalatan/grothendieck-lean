\subsection{A spectral Stein certificate below the moment-only floor}
\label{sec:independent-stein-36e-8}

\begin{theorem}[Independent sector, \st{P,C}]
\label{thm:independent-stein-36e-8}
For the original constants
\[
 a=\frac{699}{1250},\qquad b=\frac{179}{1250},\qquad
 p=\frac{2500}{1891},\qquad L=z_1(a+b|z_1|^2),
\]
every genuine complex bidegree-\((2,1)\) Wick cubic
$q\in\Hc21(z_2,\ldots,z_n)$ satisfies
\[
 \boxed{\Lambda(q)<\frac{1373953883}{10^9}
       =1.373953883<S+\frac9{25000000}
       =S+3.6\cdot10^{-7}.}
\]
This holds in every finite dimension, with arbitrary complex coefficients
and no variance or influence restriction.
\end{theorem}

\noindent
\st{P} The proof uses the inherited sixth-moment coefficient
$C=4045313/100000$, the radial Stein identity, the separate spectral
budgets, and covariance order. It does not require the additional
contraction-Gram budget. It therefore also bounds the joint-law
relaxation retaining all these constraints. It does not establish
the exact independent-sector conjecture or a bound for arbitrary
cubics involving $z_1$.

\paragraph{Normalization and the cut \st{P}.}
For $v=\E|q|^2>0$, put $Q=q/\sqrt v$, $U=|Q|^2$ and
$x=\sqrt{\E U^2-2}$. Let $X_c,r,P,Q_c$ denote the fluctuation
variables and covariances, so that
\[
 \E X_c=0,\quad 2P+Q_c=x^2,\quad
 \E X_c^2\le2P/3,\quad \E r^2\le2Q_c/3,\quad Q_c\le x^2/3.
\]
The scalar $Q_c$ is distinct from the polynomial $Q$.
The inherited moment bounds are
\[
 0\le x^2\le60,\qquad
 \E U^3\le6+9x^2+Cx^3.
\]
These moment inputs are used only for genuine $\Hc21$ cubics,
not arbitrary proper pairs of real cubic chaoses.

Let $\phi$ be continuous piecewise affine and eventually constant, and set
\[
 \Phi(u)=\int_0^u\phi(s)\,ds,\quad
 d(u)=\phi(u)-\Phi(u)/u,\quad A_\phi(u)=\Phi(u)-\phi(u),
\]
where $d(0)=0$. The test $\Phi(u)/u$ is $C^1$, with itself, its derivative,
and $u$ times its derivative bounded. The radial Stein identity gives
\[
 \E A_\phi(U)=\E[X_c\phi(U)+r\,d(U)].
\]
For $\lambda,\mu>0$, $\nu\ge0$, and $t_0,\beta_0\in\R$, define
\[
 f_0=-\tfrac12(\phi+\beta_0)+(2t_0+\lambda/3)u,\qquad
 g_0=-\tfrac12d+(t_0+\mu/3-\nu/2)u,
\]
\[
 G=f_0^2/\lambda+g_0^2/\mu+A_\phi
                       +(\nu/3-2t_0)(u^2-2).
\]
Expansion proves
\begin{align*}
 \E G(U)={}&
 \E\!\left[\lambda(X_c-f_0(U)/\lambda)^2
                +\mu(r-g_0(U)/\mu)^2\right]\\
 &+\lambda(2P/3-\E X_c^2)
  +\mu(2Q_c/3-\E r^2)+\nu(x^2/3-Q_c)\ge0.
\end{align*}
All terms are integrable by the fourth moment and the spectral budgets.
Consequently, if a cubic $R(u)=\sum_{j=0}^3R_ju^j$ satisfies
$R_3\ge0$ and $R(u)\ge h(vu)+G(u)$ on $[0,\infty)$, then
\begin{equation}\label{eq:independent-stein-node}
 \Lambda(q)\le R_0+R_1+(2+x^2)R_2+
                        (6+9x^2+Cx^3)R_3-pv.
\end{equation}
For fixed $R$, its right side is $c_0+c_2x^2+c_3x^3$ with $c_3\ge0$.
Its derivative is $x(2c_2+3c_3x)$, so an interior critical point at
$x>0$ is a minimum. Hence an interval in $x$ is bounded by its endpoints.

\paragraph{A uniform curvature estimate \st{P}.}
For $A=a+bt$, $B=a+3bt$, the squared-radius density of $L$ is
$\eta(s)=e^{-t}/(AB)$, where $s=tA^2$ and $ds=AB\,dt$.
Its complete monotonicity gives a Gaussian variance mixture
$L\stackrel{\rm law}=\sqrt D\,W$.
The square-root integral formula gives
\[
 h_D(s)-h_D(0)=\frac1{\sqrt\pi}\int_0^\infty
       \frac{1-e^{-ts/(1+Dt)}}{(1+Dt)t^{3/2}}\,dt.
\]
Thus $-h_D''(0)=\sqrt\pi D^{-3/2}/8$. Tonelli and the density formula give
\begin{align*}
 \sqrt\pi\,\E D^{-3/2}
 &=\int_0^\infty\frac{-\eta'(s)}{\sqrt s}\,ds\\
 &=\int_0^\infty e^{-t}t^{-1/2}
      (A^{-2}B^{-1}+bA^{-3}B^{-1}+3bA^{-2}B^{-2})\,dt\\
 &\le\sqrt\pi(a^{-3}+4ba^{-4}).
\end{align*}
This proves the required integrability at zero. Since $h''$ increases,
\[
 0<-h''(s)\le-h''(0)
 \le\frac{\sqrt\pi}{8}(a^{-3}+4ba^{-4})
 <\frac14(a^{-3}+4ba^{-4})<3.
\]
The last inequality is checked in rational arithmetic; $\pi<4$ is elementary.
