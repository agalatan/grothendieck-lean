\subsection{Statements and independence}
Throughout,
\[
 a=\frac{699}{1250},\qquad b=\frac{179}{1250},\qquad
 p=\varepsilon^{-1}=\frac{2500}{1891},\qquad
 L=Z(a+b|Z|^2),
\]
\[
 \Lambda(G)=\E|L+G|+\E|L-G|-p\E|G|^2,\qquad
 S=\sup_{v\ge0}\{2\E|L+\sqrt vU|-pv\}.
\]
All base Gaussian coordinates are standard proper complex Gaussians.
A \emph{fresh buffer} $U$ is independent of $Z$ and of every coordinate
used by the cubic. This independence is required in every buffered
statement below.

\begin{theorem}[The exact one-driver half-line, \st{P,C}]
\label{thm:exact-wick-halfline}
Let $X,Z,U$ be mutually independent, and put
\[
 G=c\frac{X(|X|^2-2)}{\sqrt2}+gU
      +\alpha(|Z|^2-1)X+\beta\frac{Z^2\overline X}{\sqrt2}.
\]
For every $c\ge0$, $g\ge3/16$, and $\alpha,\beta\in\C$,
\[
 \boxed{\Lambda(G)\le S.}
\]
The independently supplied strict estimate
$\Lambda(G)\le S-1/10000$ when $c\ge1/10$ also replays successfully.
An arbitrary phase of the Wick coefficient can be absorbed into $X$.
\end{theorem}

This removes the slack on the supplied half-line, but not its buffer
restriction. The proof does not assume global K or use an unproved
comparison for an arbitrary shared cubic.

\begin{theorem}[All coupled cubics with a fresh buffer, \st{P,C}]
\label{thm:all-cubic-buffer}
For every finite-dimensional $Q\in H_{2,1}$, with arbitrary dependence
on the visible coordinate and arbitrary shared drivers,
\[
 \Lambda(Q+gU)\le S-\delta\|Q\|_2^2
\]
for each of the following pairs:
\[
\begin{array}{c|c}
g\ge & \delta\\ \hline
477/1000&1/2000\\
12/25&1/200\\
1/2&3/80
\end{array}
\]
There are no amplitude, kurtosis, or dimension restrictions on $Q$.
\end{theorem}

\begin{theorem}[Sector information, \st{P,C}]
\label{thm:sector-buffer}
Let $q$ be the remote projection of $Q$, and put $Q_{\rm vis}=Q-q$.
Thus $q$ is the conditional expectation given all coordinates except $Z$.
If $q=0$, then
\[
 \begin{split}
 g\ge1/6&\ \Longrightarrow\
       \Lambda(Q+gU)\le S-\|Q\|_2^2/200,\\
 g\ge3/16&\ \Longrightarrow\
       \Lambda(Q+gU)\le S-\|Q\|_2^2/16.
 \end{split}
\]
If instead $\|q\|_2^2\le1/15$ and $g\ge2/5$, then
\[
 \Lambda(Q+gU)<S-\frac1{1000}-\frac3{10}\|Q_{\rm vis}\|_2^2.
\]
All visible $A,B,Y,W,\nu$ sectors and their interactions are allowed.
\end{theorem}

\subsection{A sharp radial Hermite contraction}
\begin{lemma}[Radial contraction, \st{P}]
\label{lem:radial-h21-contraction}
Let $T=|Z|^2$ and let $k:[0,\infty)\to[0,\infty)$ be decreasing and
measurable. For every $Q\in H_{2,1}$,
\[
 \boxed{\E[k(T)|Q|^2]\le \E k(T)\,\|Q\|_2^2.}
\]
The coefficient is sharp when a nonzero remote cubic is available.
\end{lemma}
\begin{proof}
The coordinate expansion is
\[
 Q=q+ZY+\overline ZW+(T-1)A+\frac{Z^2}{\sqrt2}B
                    +\nu\frac{Z(T-2)}{\sqrt2},
\]
where the remote bidegrees of $q,Y,W,A,B$ are respectively
$(2,1),(1,1),(2,0),(1,0),(0,1)$. Remote Hermite orthogonality,
together with the phase average of $Z$, removes every weighted cross
term. The six squared norms are consequently weighted by
\[
 1,\quad T,\quad T,\quad (T-1)^2,\quad T^2/2,\quad T(T-2)^2/2.
 \tag{43.1}
\]
Each of these polynomials $P$ has exponential mean one. Direct
integration gives
\[
 \int_x^\infty P(t)e^{-t}\,dt=e^{-x}(1+R_P(x)),
\]
where
\[
\begin{array}{c|c}
P(t)&R_P(x)\\ \hline
1&0\\
t&x\\
(t-1)^2&x^2\\
t^2/2&x+x^2/2\\
t(t-2)^2/2&x(x^2-x+2)/2 .
\end{array}
\]
All remainders are nonnegative; in the last row,
$x^2-x+2=(x-1/2)^2+7/4$. Thus each weighted probability law
stochastically dominates the exponential law. A decreasing $k$ has a
smaller expectation under each weighted law. Sum the resulting six
inequalities. Layer cake proves the assertion also for general
nonnegative decreasing measurable $k$.
\end{proof}

The bidegree constraint enters precisely here. The inadmissible
replacement $(T-2)^2/2$ instead has remainder $x(x-2)/2$, which is
negative on $(0,2)$. The contraction does not extend to that profile.

\subsection{The two-sign tangent and the scalar certificates}
Define
\[
 \rho_g(u)=\E|\sqrt u+gU|,\qquad
 K_g(u)=2\rho'_g(u)
 =\frac1{g\sqrt\pi}\int_0^1e^{-ut/g^2}
                    \sqrt{\frac{1-t}{t}}\,dt.
\]
The positive integral shows that $\rho_g$ is concave in $u$.
For arbitrary complex $F,D$, apply its tangent separately to
$|F+D|^2$ and $|F-D|^2$:
\[
 \rho_g(|F+D|^2)+\rho_g(|F-D|^2)
 \le2\rho_g(|F|^2)+K_g(|F|^2)|D|^2.
 \tag{43.2}
\]
Both signs must be retained: a single-sign substitution would leave a
nonzero diagonal gradient. In (43.2) every linear term cancels.

Put
\[
 F(g)=\Lambda(gU),\qquad
 \kappa_0(g)=\E_T K_g(T(a+bT)^2).
\]
Since $T\mapsto K_g(T(a+bT)^2)$ is decreasing,
Lemma~\ref{lem:radial-h21-contraction} gives
\[
 \boxed{\Lambda(Q+gU)\le
        F(g)+(\kappa_0(g)-p)\|Q\|_2^2.}                \tag{43.3}
\]
Moreover
\[
 K_g(u)=\frac2{\sqrt\pi}
       \int_0^{1/g}e^{-ux^2}\sqrt{1-g^2x^2}\,dx
\]
is decreasing in $g$. The scalar certificates
$\kappa_0(477/1000)<p-1/2000$,
$\kappa_0(12/25)<p-1/200$, and
$\kappa_0(1/2)<p-3/80$ therefore prove
Theorem~\ref{thm:all-cubic-buffer}.

For Theorem~\ref{thm:sector-buffer}, retain the five individual
coefficients $\kappa_P(g)=\E[P(T)K_g(T(a+bT)^2)]$ in (43.1).
The directed computation proves:
\begin{itemize}
\item at $g=1/6$, every visible coefficient is less than $p-1/200$;
\item at $g=3/16$, every visible coefficient is less than $p-1/16$;
\item at $g=2/5$, $\kappa_0<1463/1000$, every visible coefficient
is less than $p-3/10$, and $F(2/5)<13613/10000$.
\end{itemize}
The baseline $F$ is decreasing for $g\ge2/5$. For example, Gaussian
heat differentiation and the decreasing inverse-Rice mean imply
\[
 \frac{d}{dv}F(\sqrt v)
 \le \frac{\sqrt\pi}{2\sqrt{a^2+v}}-p<0
       \quad(v\ge4/25).
\]
Finally $S\ge2\E|L|=\sqrt\pi(a+3b/2)$; the rational lower
bound $\sqrt\pi>177245/100000$ already suffices to check
\[
 \sqrt\pi(a+3b/2)
 -\frac{13613}{10000}
 -\left(\frac{1463}{1000}-p\right)\frac1{15}>\frac1{1000}.
\]
This proves the claimed remote-variance window without a stored decimal
benchmark.

\subsection{Why the half-line slack can be removed}
The remainder of the proof concerns the one-driver model in
Theorem~\ref{thm:exact-wick-halfline}.
At an anchor with Wick amplitude $c_i$ and zero couplings, put
\[
 H_c=X(|X|^2-2)/\sqrt2,\quad H_A=(|Z|^2-1)X,\quad
 H_B=Z^2\overline X/\sqrt2,\quad F_i=L+c_iH_c.
\]
The inherited Rice tangent gives, for $h=c-c_i$ and
$\delta=(h,\alpha,\beta)$,
\[
 \Lambda(c,\alpha,\beta,g)\le
 \Lambda(c_i,0,0,g)+2\Re(d_i\cdot\delta)+\delta^*M_i\delta,
 \tag{43.4}
\]
\[
 (M_i)_{jk}=\E[K_g(|F_i|^2)\overline H_jH_k]-p\,1_{j=k},
 \quad
 (d_i)_j=\E[K_g(|F_i|^2)\overline F_iH_j]-p(c_i,0,0)_j.
\]
These coefficients are real at the real anchor. They include the visible
average before completing squares. Formula (43.4) holds at every
complex coupling amplitude.

 \max\{P_{i,\gamma}(c_{\rm left}),
       P_{i,\gamma}(c_{\rm right})\}
 +\max(0,-\overline M_{cc})\Delta^2/4.                \tag{43.10}

\paragraph{A further exact identity, \st{P}.}
The radial argument also has an all-degree balanced analogue. For the
standard Laguerre polynomials,
\[
 \int_x^\infty e^{-t}L_n(t)^2\,dt
 =e^{-x}\left(1+x^2\sum_{j=0}^{n-1}
                  \frac{L_j^{(1)}(x)^2}{(j+1)^2}\right).
\]
To prove it, set $A=L_{n-1}^{(1)}$ and use the coefficient identities
$nL_{n-1}=A+xA'$ and $nL_n=(1-x)A+xA'$. They give
\[
 e^{-x}(L_n^2-L_{n-1}^2)
 =-\frac{d}{dx}\left(e^{-x}\frac{x^2A^2}{n^2}\right).
\]
Integrate and telescope. Thus every normalized balanced one-coordinate
Hermite squared-modulus profile stochastically dominates the exponential
law. The proof holds for all degrees; the bundled degree-30 exact
polynomial check is a transcription audit, not an induction substitute.
No assertion about all nonzero angular frequencies is inferred from it.
