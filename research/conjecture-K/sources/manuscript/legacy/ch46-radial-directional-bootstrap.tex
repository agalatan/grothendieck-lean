\subsection{A sharper directional radial envelope}
\paragraph{Status [P]; scalar constants require their separate replay.}
For a unit planar vector $e$, let
$K_e(w)=(|w|^2-(w\cdot e)^2)/|w|^3$.
If $f\ge0$ is nonincreasing, then
\begin{equation}
 \boxed{\sup_{u,e}\int_{\mathbb R^2}
 \frac{f(|w|^2)}{\pi}K_e(w-u)\,dw
 =\int_0^\infty f(r^2)\,dr.}
 \label{eq:directional-radial-sharp}
\end{equation}
Equality holds at $u=0$; no normalization of the density is required.

For the proof, use disk layers and rotate $e=(1,0)$.
The function $y^2(x^2+y^2)^{-3/2}$ is even and decreasing in $|x|$,
so the parallel displacement of each horizontal disk chord is maximized
at zero. The remaining perpendicular displacement $u$ gives
\[
 \int_{\operatorname{disk}((0,u),R)}K_e(w)\,dw
 =2R\int_{-1}^1
 \frac{\sqrt{1-t^2}}{\sqrt{1+(u/R)^2+2(u/R)t}}\,dt
 =\pi R\,\mathbb E_{\omega\in S^3}|(u/R)e_1-\omega|^{-1}
 \le\pi R.
\]
For completeness, $\phi_\delta(x)=(|x|^2+\delta^2)^{-1/2}$ on
$\mathbb R^4$ has
$\Delta\phi_\delta=-(|x|^2+4\delta^2)/(|x|^2+\delta^2)^{5/2}<0$.
Its spherical average $U_\delta(r)$ satisfies
$(r^3U_\delta')'\le0$, hence
$U_\delta(r)\le U_\delta(0)=(1+\delta^2)^{-1/2}$.
Monotone convergence gives the asserted sphere bound. Layer cake
proves \eqref{eq:directional-radial-sharp}.

Let $k_0=\sqrt{3ab}$ and $0\le h<k_0$. Define
\[
 A_h(r)=r(a+br^2)+hr^2,\quad
 b_h(r)=\frac{(r^2-1)^2e^{-r^2}}
 {(a+br^2)(a+3br^2-2hr)},\quad
 B_h(r)=\sup_{t\ge r}b_h(t).
\]
For $F_\zeta(z)=z(a+b|z|^2)+\zeta(|z|^2-1)$, $|\zeta|=h$,
the global diffeomorphism and its Jacobian lower bound imply that
the pushforward of $(|z|^2-1)^2d\gamma(z)$ under $F_\zeta+\zeta$
has density bounded by
$B_h(A_h^{-1}(|w|))/\pi$.
Indeed $|F_\zeta(z)+\zeta|\le A_h(|z|)$ and the inverse radius is
therefore at least $A_h^{-1}(|w|)$.
Applying \eqref{eq:directional-radial-sharp} gives the uniform estimate
\[
 \mathbb E_Z(|Z|^2-1)^2K_e(F_\zeta(Z)+q)\le C(h),\qquad
 C(h)=\int_0^\infty B_h(r)(a+3br^2+2hr)\,dr.
\]
This holds for every center $q$ and direction $e$.
The function $C$ is increasing and convex: each $b_h(r)$ is increasing
convex in $h$, suprema preserve both properties, and multiplication by
the positive increasing affine factor preserves convexity.

Let $q$ be arbitrary square integrable and independent of $Z$, and let
$r_A$ be any remote Gaussian linear form of norm $u$, with arbitrary
dependence between $q$ and $r_A$. Choose $u\sqrt T<k_0$ and put
\[
 \overline{\mathcal C}(u,T)=
 \frac{2C(0)[1-(T+1)e^{-T}]
 +\int_0^Tte^{-t}C(u\sqrt t)\,dt}{3}.
\]
Convexity gives
$2\int_0^1(1-\lambda)C(\lambda h)\,d\lambda
\le[2C(0)+C(h)]/3$.
For every $\kappa<p-\overline{\mathcal C}(u,T)$, second-order integration
on $|r_A|^2\le u^2T$ and a triangle bound on its complement give
\[
 \begin{split}
 \Lambda(q+(|Z|^2-1)r_A)\le {}&\Lambda(q)
 +2\Re\mathbb E\left[(|Z|^2-1)r_A
       \frac{\overline{L+q}}{|L+q|}\right]-\kappa u^2\\
 &+\frac{9\Gamma_T^2}
 {4[p-\overline{\mathcal C}(u,T)-\kappa]},
 \qquad
 \Gamma_T\le\left(\sqrt T+\frac1{2\sqrt T}\right)e^{-T}.
 \end{split}
\]
The bad-event tangent remainder is at most $3u\Gamma_T$.
With a fresh Gaussian buffer $gY$ inside $q$, independent of all the
other variables, the error is zero if
\[
 \overline{\mathcal C}(u,T)+(T+1)e^{-T}/g\le p-\kappa.
\]
The numerical certificate evaluates only the scalar radial envelopes
and these explicit comparisons. The result does not allow a visible
B term inside the baseline $q$; the simultaneous A/B content remains
a separate obligation.

\subsection{A Jensen bootstrap for the Gaussian benchmark [P,C]}

\[

\]

\begin{equation}\label{eq:shift-baseline-jensen}
 F'(v)+p\ge\frac A{\sqrt{1+2Cv/A}},\qquad
 \sup_{v\ge0}F(v)-F(0)\ge\frac{(p+D)D^2}{2Cp}.
\end{equation}
Indeed,
$F'(v)+p=(\sqrt\pi/2)\mathbb E(V+v)^{-1/2}$.

$x\mapsto(1+vx)^{-1/2}$ proves the first bound. Integrate and evaluate at
$v=A((A/p)^2-1)/(2C)$ to obtain the second. This evaluation point need
not be the actual optimizer.

\subsection{Transport of the actual shared B content, without a tail error}
\paragraph{Status and scope.} The analytic argument below is \textup{[P]}; its
scalar constants and rectangle comparisons are \textup{[C]}. All visible
parameters are at the peak. The inverse-distance layer estimate
$C_{AB}<19/25$, the shifted B-cost, and the coherent A theorem are
explicit inputs. No Grothendieck endpoint follows without the full cover.

Let $q$ be any square-integrable remote variable. Let $r_A$ and $\bar R$
be Gaussian linear forms of the remote proper Gaussian vector; their
dependence on $q$ is unrestricted. Set $u=\|r_A\|_2$, $\sigma=\|R\|_2$,
and define
\[
 \mathcal B_A=\E\left[\frac{Z^2R}{\sqrt2}
  \frac{\overline{L+q+(T-1)r_A}}{|L+q+(T-1)r_A|}\right],\qquad
 \mathcal B_0=\E\left[\frac{Z^2R}{\sqrt2}
  \frac{\overline{L+q}}{|L+q|}\right].
\]
Here $Z$ is independent of all remote variables and $T=|Z|^2$.
For $0\le u\le1/16$,
\begin{equation}\label{eq:actual-content-transport}
 \boxed{|\mathcal B_A-\mathcal B_0|\le\frac45u\sigma.}
\end{equation}
There is no additive remainder and no positive-buffer assumption.

\paragraph{Proof.} Write $r_A=uX$, $t=|X|^2$, and truncate at
$T_u=1/(16u^2)\ge16$. Thus $u\sqrt{T_u}=1/4<k_0=\sqrt{3ab}$.
The shifted visible map has Jacobian bounded below by
$(1-|h|/k_0)P'(s)$. Differentiating the unit phase along the A segment
and applying the AB layer bound gives, on $t\le T_u$,
\[
 \frac{19}{25}u\int_0^1\E\left[
 \frac{|XR|\mathbf1_{t\le T_u}}{\sqrt{1-\lambda u\sqrt t/k_0}}
 \right]d\lambda.
\]
Gaussian regression gives
$\E(|R|^2\mid X)=\sigma^2(\rho^2t+1-\rho^2)$, so for $f\ge0$,
\[
 \E|XR|f(t)\le\sigma\max\{\E f(t),\E tf(t)\}.
\]
Define
\[
 J_j=\int_0^{16}\frac{t^je^{-t}}{\sqrt{1-\sqrt t/(16k_0)}}dt,
 \quad F_{\max}=(1-1/(4k_0))^{-1/2},\quad
 J_* =\max(J_0,J_1)+17F_{\max}e^{-16}.
\]
The extra term bounds the part $16<t\le T_u$. Convexity in $\lambda$
then bounds the good contribution by
$\frac{19}{50}(1+J_*)u\sigma$.
On the bad event the difference of unit phases is at most $2$.
Writing $\Gamma_T=\int_T^\infty\sqrt t e^{-t}dt$, regression gives
\[
 |\hbox{bad contribution}|\le\sqrt2\sigma
 \sqrt{\Gamma_{T_u}^2+e^{-2T_u}},\qquad
 \Gamma_T\le\left(\sqrt T+\frac1{2\sqrt T}\right)e^{-T}.
\]
After division by $u\sigma$, this is at most
$4\sqrt2e^{-T_u}\sqrt{T_u^2+2T_u+1/4}$, a decreasing function for
$T_u\ge16$. The exact checker proves
\[
 \frac{19}{50}(1+J_*)+2\sqrt{2306}e^{-16}<\frac45.
\]
This proves \eqref{eq:actual-content-transport}. Smooth approximation
justifies differentiation of the phase: on the good event the
pushforward densities are locally bounded and the inverse-distance
kernel is locally integrable.

\subsection{Dimension-free directional content transport [P,C]}\label{sec:directional-transport-dimfree}
