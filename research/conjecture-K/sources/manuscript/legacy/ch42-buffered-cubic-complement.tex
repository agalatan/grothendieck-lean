\subsection{The global buffered tangent and the anchor}
Write $\psi_g(z)=\E_U|z+gU|=\rho_g(|z|^2)$. The independently supplied Rice derivative has the positive representation
\[
 K_g(u):=2\rho'_g(u)=\frac1{g\sqrt\pi}\int_0^1
 e^{-ut/g^2}\sqrt{\frac{1-t}{t}}\,dt.
\]
It is decreasing, so $\rho_g$ is concave. Consequently, for all complex $F,D$,
\begin{equation}
 2\psi_g(F+D)\le2\psi_g(F)+2K_g(|F|^2)\Re(\overline FD)+K_g(|F|^2)|D|^2.
 \label{eq:rice-global}
\end{equation}
This inequality has no restriction on $D$. It averages the independent buffer first and does not assert fibrewise Gaussian domination for a cubic law.

Set
\[
 c_0=3/32,\quad \alpha_0=1/128,\quad O_0=c_0H_c+\alpha_0H_A,\quad F=L+O_0.
\]

\begin{equation}
 \Lambda(O_0+gU)\le S-\frac1{4000}-\frac{\alpha_0^2}{200}.
 \label{eq:ch42-anchor}
\end{equation}
The archived preceding certificate is an explicit dependency. This continuation does not estimate $S$ numerically or claim that its anchor input follows from the new coefficient integrals alone.

For $h=c-c_0$, $\xi=\alpha-\alpha_0$, $\zeta=\beta$, the odd perturbation is $D_o=hH_c+\xi H_A+\zeta H_B$. Averaging~\eqref{eq:rice-global} and subtracting the price yields

\subsection{The simultaneous interaction identity}
\begin{lemma}[Odd/even cancellation about the fixed anchor, \st{P}]
Let $O=cH_c+\alpha H_A+\beta H_B$ and $E=yH_Y+wH_W+\nu H_\nu$. Decompose $J=J_o+J_e$ into odd and even parts under $(X,V)\mapsto(-X,-V)$, with $Z$ fixed. Then the original two-sign payoff equals
\[
 \E\{\psi_g(L+O+J_o+E+J_e)+\psi_g(L+O+J_o-E-J_e)\}.
\]
Writing $D=O-O_0+J_o$ and $B=E+J_e$, it is at most
\begin{equation}
 2\E\psi_g(F)+2\E[K_g(|F|^2)\Re(\overline FD)]
       +\E[K_g(|F|^2)(|D|^2+|B|^2)].
 \label{eq:ch42-parity}
\end{equation}
\end{lemma}
\begin{proof}
Apply the measure-preserving half-turn to the second term of the original expectation. The odd part changes sign and the even part does not. Apply half of~\eqref{eq:rice-global} to $D+B$ and $D-B$, and add. The even linear terms and all odd/even quadratic cross terms cancel by
$|D+B|^2+|D-B|^2=2|D|^2+2|B|^2$.
\end{proof}
Both $J_o,J_e$ stay in $H_{2,1}$, have zero conditional mean given $Z,X$, and are orthogonal in unweighted $L^2$. Since $F,K_g(|F|^2)$ are measurable in $Z,X$, conditional centering removes the remaining linear $J_o$ term and the cross terms with $O-O_0$ and $E$, including the price cross terms. Thus, in addition to the odd quadratic majorant, only
\begin{equation}
 \E[K_g(|F|^2)|E|^2]-p\|E\|_2^2
 +\E[K_g(|F|^2)(|J_o|^2+|J_e|^2)]-p\|J\|_2^2
 \label{eq:ch42-extra}
\end{equation}
remains. This cancellation holds at every target amplitude and phase. No conditional orthogonality of the target $A/B$ and even sectors is assumed.
