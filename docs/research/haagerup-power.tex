\section{A global upper bound from a canonical resolvent completion}
For a real polynomial, let $\|P\|_{\mathcal A}$ denote its absolute
coefficient sum. Haagerup's Gaussian phase kernel is
\[
 h(z)=\frac\pi4\sum_{k\ge0}r_kz|z|^{2k},\qquad
 r_0=1,\quad r_{k+1}=r_k\frac{(2k+1)^2}{4(k+1)(k+2)}.
\]
The Gaussian identity, tensor realization, complete residual estimate, and
matrix transfer are proved in Lean. The resulting bound has no external
analytic or numerical premises:
\[
 \boxed{K_G^{\mathbb C}\le\frac{1404909140}{1000000000}=1.404909140.}
\]

\subsection{Complete monotonicity of the coefficients}
Write $(\Delta a)_k=a_k-a_{k+1}$. A sequence is completely monotone when
$\Delta^j a_k\ge0$ for all $j,k\ge0$. Every difference sequence is then
nonnegative and decreasing, and
$0\le\Delta^j a_k\le a_k$. Thus $a_k\to0$ implies
$\Delta^j a_k\to0$ for every fixed $j$.

For $0\le\alpha\le\beta$ and $\beta>0$, put
\[
 u(\alpha,\beta)_k=\frac{(\alpha)_k}{(\beta)_k},
 \qquad (\alpha)_k=\prod_{i<k}(\alpha+i).
\]
The recurrence for this quotient gives
\[
 \Delta u(\alpha,\beta)
 =\frac{\beta-\alpha}{\beta}\,u(\alpha,\beta+1).
\]
Induction on the difference order proves complete monotonicity.
The discrete product rule
\[
 \Delta(uv)_k=(\Delta u_k)v_k+u_{k+1}\Delta v_k
\]
shows that the product of two completely monotone sequences is completely
monotone: differences, shifts, addition, and multiplication by nonnegative
constants preserve the property. The Haagerup recurrence identifies
\[
 r_k=u(1/2,1)_k\,u(1/2,2)_k,
\]
so its coefficients are completely monotone. Their summability also gives
$r_k\to0$. These arguments hold at every difference order.

\subsection{A finite inverse certificate and a uniform Abel estimate}
Let $A,T,R$ be real polynomials, with $\|A\|_{\mathcal A}\le1$, and set
$E=1-R(1-A)$. Suppose
\[
 \|E\|_{\mathcal A}\le\eta<1,
 \qquad \|RT\|_{\mathcal A}\le B,
 \qquad \|RA\|_{\mathcal A}\le C.
\]
For every polynomial $S$, the identity $S=ES+R(1-A)S$ gives
\[
 (1-\eta)\|S\|_{\mathcal A}\le\|R(1-A)S\|_{\mathcal A}.
\]
Apply this to $S_n=\sum_{i<n}TA^i$. Since
$(1-A)S_n=T-TA^n$, submultiplicativity gives
\[
 \|S_n\|_{\mathcal A}\le M,\qquad M=\frac{2B}{1-\eta}.
\]
For decreasing nonnegative weights $b_i$, finite summation by parts yields
\[
 \left\|\sum_{i<n}b_iTA^i\right\|_{\mathcal A}\le b_0M.
\]
Indeed, express the weighted sum as its last weight times $S_n$ plus
$\sum_{i<n-1}(b_i-b_{i+1})S_{i+1}$; the scalar weights telescope to $b_0$.

\subsection{The canonical completion at every order}
For a completely monotone sequence $a$, write
\[
 W_n(a)=\sum_{i<n}a_{K+i}TA^i,\qquad
 Q_m(a)=\sum_{j<m}(\Delta^j a)_K(-RA)^jRT.
\]
One summation by parts gives
\[
 (1-A)W_n(a)=a_KT-AW_n(\Delta a)-a_{K+n}TA^n.
\]
Multiplication by $R$ and the definition of $E$ give the recurrence
\[
 W_n(a)=a_KRT-RA\,W_n(\Delta a)+E W_n(a)-a_{K+n}RT A^n.
\]
Induction on $m$ proves the exact finite identity
\begin{align*}
 W_n(a)-Q_m(a)
 ={}&(-RA)^mW_n(\Delta^m a)
   +E\sum_{j<m}(-RA)^jW_n(\Delta^j a)\\
 &-\sum_{j<m}(\Delta^j a)_{K+n}(-RA)^jRT A^n.
\end{align*}
Applying the uniform Abel estimate at each difference order yields
\begin{align*}
 \|W_n(a)-Q_m(a)\|_{\mathcal A}
 \le{}&M\left[C^m(\Delta^m a)_K+
       \eta\sum_{j<m}C^j(\Delta^j a)_K\right]\\
 &+B\sum_{j<m}C^j(\Delta^j a)_{K+n}.
\end{align*}

Now take $A=XP^2$ and $T=X^KP^{2K+1}$, with
$\|P\|_{\mathcal A}\le1$ and $a_k\to0$.
Each fixed coefficient prefix of the composed series is determined by
finitely many terms, since the term indexed by $k$ contains $X^k$.
Fix that prefix and let $n\to\infty$. Every boundary weight vanishes,
and the bound is independent of the prefix. Taking the supremum over
finite absolute coefficient sums gives
\[
 \left\|\sum_{k\ge K}a_kX^kP^{2k+1}-Q_m(a)\right\|_{\mathcal A}
 \le\frac{2B}{1-\eta}
 \left[C^m(\Delta^m a)_K+
       \eta\sum_{j<m}C^j(\Delta^j a)_K\right].
\]
Here the series and its norm are defined coefficientwise. The proof uses
finite polynomial identities and coefficient locality, without constructing
an infinite polynomial inverse or interchanging two series.
